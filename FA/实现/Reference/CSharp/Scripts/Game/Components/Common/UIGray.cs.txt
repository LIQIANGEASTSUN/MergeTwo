using UnityEngine;
using UnityEngine.UI;

[XLua.LuaCallCSharp]
public class UIGray : MonoBehaviour
{
    public Material grayMaterial;

    private Graphic[] grapihcArr;
    private bool inited = false;

    private void Start()
    {
        Init();
        SetGray(true);
    }

    private void Init()
    {
        grapihcArr = gameObject.GetComponentsInChildren<Graphic>(true);
        if (grayMaterial == null)
        {
            grayMaterial = new Material(Shader.Find("HomeLand/Default-UI-Gray"));
        }
        inited = true;
    }

    private void OnEnable()
    {
        SetGray(true);
    }

    private void OnDisable()
    {
        SetGray(false);
    }

    private void SetGray(bool isGray)
    {
        if (!inited)
            Init();

        for (int i = 0; i < grapihcArr.Length; i++)
        {
            if (isGray)
            {
                grapihcArr[i].material = grayMaterial;
            }
            else
            {
                grapihcArr[i].material = grapihcArr[i].defaultMaterial;
            }
        }
    }
}
