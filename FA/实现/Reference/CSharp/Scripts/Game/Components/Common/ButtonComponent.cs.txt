using DG.Tweening;
using UnityEngine;
using UnityEngine.EventSystems;
using UnityEngine.UI;

namespace BetaGame
{
    [XLua.LuaCallCSharp]
    public class ButtonComponent : Button
    {
        private Material material;
        //private UIEventListener eventListener;
        private Graphic[] graphics;
        private Color highLightColor = new Color(0.08f, 0.08f, 0.08f, 1);
        //private float scaleSize = 0.9f;
        //private float animTime = 0.1f;
        private Animator mAnimator;
        [Tooltip("保持原动画,避免被强制添加缩放动画")]
        public bool keepTransation = false;

        public bool Interactable
        {
            get
            {
                return interactable;
            }
            set
            {
                interactable = value;
                SetButtonGray(value);
            }
        }

        protected override void Awake()
        {
            base.Awake();
            if (Application.isPlaying)
            {
                Init();
                var navigation = this.navigation;
                navigation.mode = Navigation.Mode.None;
                this.navigation = navigation;
            }
        }

        private void Init()
        {
            if (material == null)
            {
                try
                {
                    var inst = Shader.Find("UI/Button");
                    if (inst == null)
                    {
                        XDebug.LogError($"Shader:{gameObject.name}");
                    }
                    material = new Material(inst);
                }
                catch (System.Exception exp)
                {
                    XDebug.LogException($"ButtonComponent:{exp}");
                }
            }
            if (graphics == null)
            {
                graphics = GetComponentsInChildren<Graphic>(true);

                for (int i = 0; i < graphics.Length; i++)
                {
                    if (graphics[i].material == graphics[i].defaultMaterial)
                    {
                        graphics[i].material = material;
                    }
                }

                mAnimator = GetComponent<Animator>();
                if (mAnimator == null && !keepTransation)
                {
                    mAnimator = gameObject.AddComponent<Animator>();
                    RuntimeAnimatorController controller = AssetLoaderUtil.GetAsset("Prefab/Buildin/Animator/Button.controller") as RuntimeAnimatorController;
                    if (controller == null)
                    {
                        XDebug.LogError("Button animator controller not loaded!");
                    }
                    mAnimator.runtimeAnimatorController = controller;
                }
                //transition = Transition.Animation;
                if (mAnimator && mAnimator.runtimeAnimatorController && mAnimator.runtimeAnimatorController.name == "Button")
                    transition = Transition.Animation;

            }
            SetButtonGray(interactable);
        }

        public void SetButtonGray(bool _interactable)
        {
            float grayValue = _interactable ? 0 : 1;
            if (graphics == null)
            {
                Init();
            }

            //material.SetFloat("_Gray", 0);
            for (int i = 0; i < graphics.Length; i++)
            {
                if (graphics[i] != null && graphics[i].material != null)
                {
                    graphics[i].material.SetFloat("_Gray", grayValue);
                }
            }
        }

        public override void OnPointerDown(PointerEventData eventData)
        {
            base.OnPointerDown(eventData);
            OnPressed();
        }

        public override void OnPointerUp(PointerEventData eventData)
        {
            base.OnPointerUp(eventData);
            OnNormal();
        }

        private void OnNormal()
        {
            if (!interactable)
            {
                return;
            }

            //Sequence seq = DOTween.Sequence();
            //seq.Append(transform.DOScale(1.1f, 4/30));
            //seq.Append(transform.DOScale(1f, 3/30));

            for (int i = 0; i < graphics.Length; i++)
            {
                if (graphics[i] != null)
                {
                    var mat = graphics[i].materialForRendering;
                    if (mat && mat.HasProperty("_Highlight"))
                        mat.DOColor(Color.black, "_HighLight", 7 / 30F);
                }
            }
        }

        private void OnPressed()
        {
            if (!interactable)
            {
                return;
            }
            //Debug.Log("Button Pressed!");
            //transform.DOScale(0.9f, 2/30);
            for (int i = 0; i < graphics.Length; i++)
            {
                if (graphics[i] != null)
                {
                    var mat = graphics[i].materialForRendering;
                    if (mat && mat.HasProperty("_Highlight"))
                        mat.DOColor(highLightColor, "_HighLight", 4 / 60F);
                }
            }
        }

#if UNITY_EDITOR
        /// <summary>
        /// To avoid default navigation set as Automatic which may cause Tab issues.
        /// Ex: when a selectable widget has been selected, Press Tab then the unity will focus on an unexpected widget!
        /// </summary>
        protected override void Reset()
        {
            base.Reset();

            var navigation = this.navigation;
            navigation.mode = Navigation.Mode.None;
            this.navigation = navigation;
        }
#endif
    }
}
