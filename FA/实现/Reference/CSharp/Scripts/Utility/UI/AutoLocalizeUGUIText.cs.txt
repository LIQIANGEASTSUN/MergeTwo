using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using BetaGame;

[RequireComponent(typeof(Text)), DisallowMultipleComponent]
public class AutoLocalizeUGUIText : MonoBehaviour
{
    public string key;
    Text text;
    
    private void Awake()
    {
        text = GetComponent<Text>();
    }

    private void OnEnable()
    {
        if (text == null || Localization.Instance == null || string.IsNullOrEmpty(key))
            return;
        SetText();
        Localization.Instance.onLanguageChanged += SetText;
    }

    private void OnDisable()
    {
        if (Localization.Instance == null)
            return;
        Localization.Instance.onLanguageChanged -= SetText;
    }

    void SetText()
    {
        text.text = Localization.Instance.GetText(key);
    }
}
